
CREATE TABLE cities (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    country VARCHAR(255),
    latitude FLOAT,
    longitude FLOAT,
    icon VARCHAR(255)
);


CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    password VARCHAR(255),
    icon VARCHAR(255),
    fav_city INT,
    email VARCHAR(255),
    username VARCHAR(255)
);


CREATE TABLE types (
    id SERIAL PRIMARY KEY,
    type VARCHAR(255),
    icon VARCHAR(255)
);


CREATE TABLE events (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(255),
    descr TEXT,
    dia DATE,
    hora TIME,
    color VARCHAR(255),
    fk_type INT,
    fk_id_user INT,
    fk_city INT,
    
  
    CONSTRAINT fk_event_type FOREIGN KEY (fk_type) REFERENCES types(id) ON DELETE SET NULL,
    CONSTRAINT fk_event_user FOREIGN KEY (fk_id_user) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT fk_event_city FOREIGN KEY (fk_city) REFERENCES cities(id) ON DELETE SET NULL
);
